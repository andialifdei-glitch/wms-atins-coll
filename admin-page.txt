"use client";

import React, { useState } from "react";
import { konterDb } from "@/lib/konterDb";

export default function AdminKonterPage() {
  const [auditLogs] = useState(konterDb.getAuditLog());
  const [namaKonter, setNamaKonter] = useState("Emir Cell");
  const [alamatKonter, setAlamatKonter] = useState("Banjaran, Bandung");
  const [pesanSimpan, setPesanSimpan] = useState<string | null>(null);

  const handleSimpanProfil = (e: React.FormEvent) => {
    e.preventDefault();
    setPesanSimpan("Profil Konter berhasil diperbarui!");
  };

  return (
    <div className="space-y-6">
      <div className="pb-4 border-b border-slate-800">
        <h1 className="text-2xl font-bold text-white tracking-tight">Administrasi & Keamanan Konter</h1>
        <p className="text-sm text-slate-400 mt-1">Pengaturan identitas toko, status isolasi sistem, dan audit trail log.</p>
      </div>

      {pesanSimpan && (
        <div className="p-3 bg-emerald-500/20 border border-emerald-500/40 text-emerald-300 rounded-lg text-sm">
          {pesanSimpan}
        </div>
      )}

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        <div className="bg-slate-800/80 border border-slate-700/80 rounded-xl p-5 shadow-sm">
          <h2 className="text-base font-semibold text-white mb-4">Profil Usaha Konter</h2>
          <form onSubmit={handleSimpanProfil} className="space-y-4">
            <div>
              <label className="block text-xs font-medium text-slate-300 mb-1">Nama Usaha / Konter</label>
              <input
                type="text"
                value={namaKonter}
                onChange={(e) => setNamaKonter(e.target.value)}
                className="w-full bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-sm text-white"
                required
              />
            </div>

            <div>
              <label className="block text-xs font-medium text-slate-300 mb-1">Alamat / Lokasi</label>
              <input
                type="text"
                value={alamatKonter}
                onChange={(e) => setAlamatKonter(e.target.value)}
                className="w-full bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-sm text-white"
                required
              />
            </div>

            <div className="pt-2">
              <div className="text-xs text-slate-400 mb-2">
                Status Runtime: <span className="text-emerald-400 font-semibold">Cloudflare Workers Compatible</span>
              </div>
              <div className="text-xs text-slate-400 mb-4">
                Proteksi WMS: <span className="text-cyan-400 font-semibold">Isolasi Penuh 100% Terverifikasi</span>
              </div>
            </div>

            <button
              type="submit"
              className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-500 text-white font-semibold rounded-lg text-sm shadow-md"
            >
              Simpan Pengaturan
            </button>
          </form>
        </div>

        <div className="lg:col-span-2 bg-slate-800/80 border border-slate-700/80 rounded-xl p-5 shadow-sm">
          <h2 className="text-base font-semibold text-white mb-4">Audit Log & Rekam Aktivitas</h2>
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm text-slate-300">
              <thead className="bg-slate-900/60 text-xs uppercase text-slate-400">
                <tr>
                  <th className="px-4 py-3">Timestamp</th>
                  <th className="px-4 py-3">Aktor</th>
                  <th className="px-4 py-3">Aksi</th>
                  <th className="px-4 py-3">Detail</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-700/60">
                {auditLogs.length === 0 ? (
                  <tr>
                    <td colSpan={4} className="px-4 py-8 text-center text-slate-400 text-sm">
                      Belum ada audit log aktivitas tercatat.
                    </td>
                  </tr>
                ) : (
                  auditLogs.map((log) => (
                    <tr key={log.id} className="hover:bg-slate-750">
                      <td className="px-4 py-3 text-xs text-slate-400 font-mono">
                        {new Date(log.timestamp).toLocaleTimeString("id-ID")}
                      </td>
                      <td className="px-4 py-3 text-xs font-semibold text-white">{(log.aktor || log.user || "System")}</td>
                      <td className="px-4 py-3">
                        <span className="px-2 py-0.5 text-[10px] rounded bg-indigo-500/20 text-indigo-300 border border-indigo-500/30">
                          {(log.aksi || log.action || "Log")}
                        </span>
                      </td>
                      <td className="px-4 py-3 text-xs text-slate-300">{(log.detail || log.details || "-")}</td>
                    </tr>
                  ))
                )}
              </tbody>
            </table>
          </div>
        </div>
      </div>
    </div>
  );
}
